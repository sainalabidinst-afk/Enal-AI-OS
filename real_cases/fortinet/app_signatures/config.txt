config application list
    edit "SOCIAL-MEDIA-BLOCK"
        set comment "Block social media during work hours"
        config entries
            edit 1
                set action block
                set category "Social.Media"
                set protocol tcp
                set behavior "on"
                set signature "Facebook"
            next
            edit 2
                set action block
                set category "Social.Media"
                set protocol tcp
                set signature "Twitter"
            next
        end
    next
end
config firewall policy
    edit 1
        set name "BLOCK-SOCIAL"
        set srcintf "internal"
        set dstintf "wan1"
        set srcaddr "all"
        set dstaddr "all"
        set action accept
        set schedule "work-hours"
        set service "ALL"
        set application-list "SOCIAL-MEDIA-BLOCK"
    next
end
