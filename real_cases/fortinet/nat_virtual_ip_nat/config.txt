config firewall policy
    edit 0
        set name "LAN-TO-WAN-NAT"
        set srcintf "internal"
        set dstintf "wan1"
        set srcaddr "LAN-subnet"
        set dstaddr "all"
        set action accept
        set schedule "always"
        set service "ALL"
        set nat enable
    next
end

config firewall address
    edit "LAN-subnet"
        set subnet 192.168.50.0 255.255.255.0
    next
end

config firewall address
    edit "DMZ-server"
        set subnet 192.168.50.100 255.255.255.255
    next
end

config firewall vip
    edit "Web-VIP"
        set extip 203.0.113.10
        set extintf "wan1"
        set mappedip "DMZ-server"
    next
end
