config wanopt profile
    edit "WAN-OPT-PROFILE"
        set comment "WAN optimization profile"
        set auth-group "default"
        set auto-detect enable
    next
end
config wanopt peer
    edit 1
        set peer-ip 10.0.1.2
        set port 7810
    next
end
config firewall policy
    edit 1
        set name "WAN-OPT-POLICY"
        set srcintf "internal"
        set dstintf "wan1"
        set action accept
        set wanopt-profile "WAN-OPT-PROFILE"
        set wanopt-peer 1
    next
end
